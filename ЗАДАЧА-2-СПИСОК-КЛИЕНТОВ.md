### 1. **server-new.js** (Backend)

Добавьте около строки 4100:

```javascript
// Получить список всех клиентов
app.get('/api/business/customers', requireBusinessAuth, (req, res) => {
  try {
    const business = getBusinessData();
    const { search } = req.query;
    
    const customersMap = new Map();
    const conversations = business.conversations || [];
    
    // Пройти по всем диалогам
    conversations.forEach(conv => {
      const customerId = conv.customerId || conv.sender_id || conv.from;
      
      if (!customerId) return;
      
      // Получить информацию о клиенте
      const customerName = conv.customerName || conv.sender_name || 'Без имени';
      const customerPhone = conv.customerPhone || conv.sender_phone || 'Не указан';
      const platform = conv.platform || 'unknown';
      
      // Получить последнее сообщение от клиента
      const customerMessages = conv.messages?.filter(m => m.from === 'customer') || [];
      const lastMessage = customerMessages[customerMessages.length - 1];
      
      // Если клиента ещё нет в Map или это более свежее сообщение
      if (!customersMap.has(customerId) || 
          (lastMessage && new Date(lastMessage.timestamp) > new Date(customersMap.get(customerId).lastMessage?.timestamp || 0))) {
        
        customersMap.set(customerId, {
          id: customerId,
          name: customerName,
          phone: customerPhone,
          platform: platform,
          lastMessage: lastMessage ? {
            text: lastMessage.text || lastMessage.body || '',
            timestamp: lastMessage.timestamp,
            from: lastMessage.from
          } : null,
          totalMessages: customerMessages.length,
          firstContact: conv.createdAt || conv.messages?.[0]?.timestamp
        });
      }
    });
    
    // Преобразовать Map в массив
    let customers = Array.from(customersMap.values());
    
    // Применить поиск если есть
    if (search) {
      const searchLower = search.toLowerCase();
      customers = customers.filter(c => 
        c.name.toLowerCase().includes(searchLower) ||
        c.phone.includes(search)
      );
    }
    
    // Отсортировать по времени последнего сообщения (новые сначала)
    customers.sort((a, b) => {
      const timeA = a.lastMessage?.timestamp || a.firstContact || 0;
      const timeB = b.lastMessage?.timestamp || b.firstContact || 0;
      return new Date(timeB) - new Date(timeA);
    });
    
    res.json({
      success: true,
      customers,
      total: customers.length
    });
    
  } catch (error) {
    console.error('Ошибка получения клиентов:', error);
    res.status(500).json({ error: error.message });
  }
});
```

## 🆘 Частые Проблемы

**Проблема**: "Cannot read property 'conversations' of undefined"  
**Решение**: Проверьте структуру businesses.json. Возможно диалоги хранятся под другим именем

**Проблема**: Дубликаты клиентов в списке  
**Решение**: Убедитесь что используете `Map` для хранения уникальных клиентов

**Проблема**: Поиск не работает  
**Решение**: Проверьте что query параметр передаётся: `/api/business/customers?search=...`

**Проблема**: Время показывается неправильно  
**Решение**: Проверьте формат timestamp. Используйте `new Date(timestamp)` для преобразования
