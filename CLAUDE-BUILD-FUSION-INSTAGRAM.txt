```javascript
function renderInstagramRule(rule, index) {
    return `
        <div class="rule-card" data-rule-index="${index}">
            <div class="rule-header">
                <h5>Правило #${index + 1}</h5>
                <button class="btn-icon" onclick="deleteInstagramRule(${index})">
                    <i data-lucide="trash-2"></i>
                </button>
            </div>
            
            <div class="rule-field">
                <label>Применить к:</label>
                <label class="checkbox-label">
                    <input type="checkbox" ${rule.scope.includes('dm') ? 'checked' : ''} 
                           onchange="updateRuleScope(${index}, 'dm', this.checked)">
                    Direct Messages (Директ)
                </label>
                <label class="checkbox-label">
                    <input type="checkbox" ${rule.scope.includes('comments') ? 'checked' : ''} 
                           onchange="updateRuleScope(${index}, 'comments', this.checked)">
                    Comments (Комментарии)
                </label>
            </div>
            
            <div class="rule-field">
                <label>Ключевые слова (оставьте пустым для обработки всех сообщений):</label>
                <div class="keyword-input-group">
                    <input type="text" 
                           class="keyword-input" 
                           id="keyword-input-${index}"
                           placeholder="Введите слово и нажмите Enter"
                           onkeypress="if(event.key==='Enter'){addKeyword(${index});event.preventDefault();}">
                </div>
                <div class="keyword-tags" id="keywords-${index}">
                    ${rule.keywords.map(kw => `
                        <span class="tag">${kw} <button onclick="removeKeyword(${index}, '${kw}')">×</button></span>
                    `).join('')}
                </div>
            </div>
            
            <div class="rule-field">
                <label>Режим ответа:</label>
                <select class="response-mode-select" onchange="updateResponseMode(${index}, this.value)">
                    <option value="ai" ${rule.mode === 'ai' ? 'selected' : ''}>
                        Allow AI to Answer Autonomously (ИИ генерирует ответ)
                    </option>
                    <option value="template" ${rule.mode === 'template' ? 'selected' : ''}>
                        Template Response (Шаблонный ответ)
                    </option>
                    <option value="append" ${rule.mode === 'append' ? 'selected' : ''}>
                        Append to Prompt (Добавить контекст к промпту)
                    </option>
                </select>
            </div>
            
            <div class="mode-template" style="display:${rule.mode === 'template' ? 'block' : 'none'};" id="template-${index}">
                <label>Текст шаблона:</label>
                <textarea onchange="updateRuleContent(${index}, this.value)" 
                          placeholder="Ваш автоматический ответ...">${rule.content || ''}</textarea>
            </div>
            
            <div class="mode-append" style="display:${rule.mode === 'append' ? 'block' : 'none'};" id="append-${index}">
                <label>Контекст для добавления к промпту:</label>
                <textarea onchange="updateRuleContent(${index}, this.value)" 
                          placeholder="Инструкции для добавления к базовому промпту ИИ...">${rule.content || ''}</textarea>
            </div>
        </div>
    `;
}
```

### STEP 4: Add Rule Management Functions

```javascript
let instagramRules = [];

function addInstagramRule() {
    const newRule = {
        id: 'rule_' + Date.now(),
        scope: ['dm'],
        keywords: [],
        mode: 'ai',
        content: ''
    };
    instagramRules.push(newRule);
    renderInstagramRules();
    saveInstagramConfig();
}

function deleteInstagramRule(index) {
    instagramRules.splice(index, 1);
    renderInstagramRules();
    saveInstagramConfig();
}

function updateRuleScope(index, scopeType, checked) {
    if (checked) {
        if (!instagramRules[index].scope.includes(scopeType)) {
            instagramRules[index].scope.push(scopeType);
        }
    } else {
        instagramRules[index].scope = instagramRules[index].scope.filter(s => s !== scopeType);
    }
    saveInstagramConfig();
}

function addKeyword(index) {
    const input = document.getElementById(`keyword-input-${index}`);
    const keyword = input.value.trim();
    if (keyword && !instagramRules[index].keywords.includes(keyword)) {
        instagramRules[index].keywords.push(keyword);
        renderInstagramRules();
        saveInstagramConfig();
    }
    input.value = '';
}

function removeKeyword(index, keyword) {
    instagramRules[index].keywords = instagramRules[index].keywords.filter(k => k !== keyword);
    renderInstagramRules();
    saveInstagramConfig();
}

function updateResponseMode(index, mode) {
    instagramRules[index].mode = mode;
    document.getElementById(`template-${index}`).style.display = mode === 'template' ? 'block' : 'none';
    document.getElementById(`append-${index}`).style.display = mode === 'append' ? 'block' : 'none';
    saveInstagramConfig();
}

function updateRuleContent(index, content) {
    instagramRules[index].content = content;
    saveInstagramConfig();
}

function renderInstagramRules() {
    const container = document.getElementById('instagram-rules-container');
    container.innerHTML = instagramRules.map((rule, i) => renderInstagramRule(rule, i)).join('');
    lucide.createIcons();
}
```

### STEP 6: Add Message Processing Logic

```javascript
function findMatchingInstagramRule(messageText, rules, isComment) {
    const scope = isComment ? 'comments' : 'dm';
    
    // First pass: keyword-triggered rules
    for (const rule of rules) {
        if (!rule.scope.includes(scope)) continue;
        
        if (rule.keywords && rule.keywords.length > 0) {
            const hasMatch = rule.keywords.some(kw => 
                messageText.toLowerCase().includes(kw.toLowerCase())
            );
            if (hasMatch) return rule;
        }
    }
    
    // Second pass: catch-all rules (no keywords)
    for (const rule of rules) {
        if (!rule.scope.includes(scope)) continue;
        if (!rule.keywords || rule.keywords.length === 0) {
            return rule;
        }
    }
    
    return null;
}

async function processInstagramMessageWithRules(businessId, message, isComment) {
    const business = readBusiness(businessId);
    const config = business.channels?.instagram || {};
    
    // Check if enabled
    if (isComment && !config.commentsEnabled) return;
    if (!isComment && !config.dmEnabled) return;
    
    // Find matching rule
    const rule = findMatchingInstagramRule(message.text, config.rules || [], isComment);
    if (!rule) return; // No rule matched
    
    let response;
    
    if (rule.mode === 'template') {
        // Instant template response
        response = rule.content;
    }
    else if (rule.mode === 'append') {
        // Append context and generate AI response
        const fullPrompt = business.aiPrompt + "\n\n" + rule.content;
        response = await generateGroqResponse(fullPrompt, message.text);
    }
    else if (rule.mode === 'ai') {
        // Standard AI response
        response = await generateGroqResponse(business.aiPrompt, message.text);
    }
    
    // Send response
    await sendInstagramReply(message.id, response, isComment);
    
    return response;
}
```
